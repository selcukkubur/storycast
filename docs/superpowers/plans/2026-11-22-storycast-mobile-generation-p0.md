# Storycast Mobile — Real Film Generation (P0) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the Expo app produce real films end-to-end through the billing worker: real job, live stage events, real MP4, refund on failure, resume after interruption, trial watermark.

**Architecture:** Copy-adapt the web studio core (`director.ts`, `pipeline.ts`) into `mobile/src/lib/studio/` behind four shims with the same module names — `fal.ts` (billing-backed run/duration/upload), `data.ts` (bundled config/films JSON), `store.ts` (AsyncStorage, injectable), `voices.ts` (pass-through checkVoice) — so the ported files keep their imports. The create flow owns the job lifecycle: start → runFilm(events) → settle. The app stays foreground-only with an honest "Keep Storycast open" guard.

**Tech Stack:** Expo SDK 58, React Native, TypeScript, NativeWind, expo-video (existing), expo-secure-store, @react-native-async-storage/async-storage, vitest (node) for lib tests, sharp (from `billing/node_modules`) for the watermark asset.

**Spec:** `docs/superpowers/specs/2026-11-22-storycast-mobile-generation-design.md`

## Global Constraints

- No changes to `web/` in this plan — the web keeps running as-is (copy-adapt, not refactor).
- Managed only: every fal call goes through the billing worker; no fal key exists in the app.
- Foreground-only generation: the copy is "Keep Storycast open"; the old "safe to leave / we'll nudge you" promise is removed everywhere until Phase 2.
- Trial = 1 credit, ≤1 min, 480P, watermark; server enforces caps (402 credits / 422 cap).
- Watermark = a 2-second branded card clip appended before the end card, only when `POST /jobs` returned `watermark: true`.
- Data ships bundled: `mobile/assets/data/{config,films}.json` copied from `web/public/data/` by `mobile/scripts/sync-data.mjs`; no network fetch for catalog.
- Config base URL: `EXPO_PUBLIC_BILLING_API` (dev = cloudflared quick tunnel to `wrangler dev`).
- Device token lives in `expo-secure-store`; on 401 the session is cleared and one re-registration is attempted.
- The 8 UI stages are: script, character, voice, keyframes, shots, score, edit, subtitles (same order as the rolling screen tiles).
- `/dev/credits` exists only when the worker env `DEV_CREDITS_TOKEN` is set; it must never be set in a deployed environment.
- Tests are node-only (`vitest`, no RN rendering). Device E2E is a manual checklist.
- Metro's file watcher is broken in this environment: assume `npx expo start --tunnel --clear` per change batch.

## Scope Notes

- **`cost.ts` is deliberately not ported.** It computes USD from fal list prices for the BYOK web app; mobile pricing is tickets (server-owned). The mobile ticket math already lives in `src/flows/shared/types.ts` (`creditsFor`) and is parity-tested against the worker in `billing/test/parity.test.ts`. Porting USD math would be dead code.
- **StoreKit stays mocked.** The paywall's confirm sheet is labeled "Demo purchase" in Task 9; real IAP is a separate future spec.
- Phase 2 (Durable Object orchestration + push notifications) is out of scope; the plan only removes the false promise.

```ts
export type UiStage =
  | "script"
  | "character"
  | "voice"
  | "keyframes"
  | "shots"
  | "score"
  | "edit"
  | "subtitles";

export const UI_STAGES: UiStage[] = [
  "script",
  "character",
  "voice",
  "keyframes",
  "shots",
  "score",
  "edit",
  "subtitles",
];

const EVENT_TO_UI: Record<string, UiStage> = {
  director: "script",
  character: "character",
  voice: "voice",
  keyframes: "keyframes",
  shots: "shots",
  music: "score",
  assemble: "edit",
  subtitles: "subtitles",
};

export function mapEventStage(stage: string): UiStage | "done" | "error" | null {
  if (stage === "done" || stage === "error") return stage;
  return EVENT_TO_UI[stage] ?? null;
}

export function stageProgress(stage: UiStage | "done" | "error" | null): number {
  if (stage === "done") return 100;
  if (!stage || stage === "error") return 0;
  return Math.round(((UI_STAGES.indexOf(stage) + 1) / UI_STAGES.length) * 100);
}
```

```ts
export type KeyValueStore = { getItem(key: string): Promise<string | null>; setItem(key: string, value: string): Promise<void> };

let storage: KeyValueStore | null = null;
export function configureStore(next: KeyValueStore) {
  storage = next;
}
async function kv(): Promise<KeyValueStore> {
  if (!storage) storage = (await import("@react-native-async-storage/async-storage")).default as KeyValueStore;
  return storage;
}

const KEY = "storycast.films.v1";

async function readAll(): Promise<FilmRecord[]> {
  try {
    return JSON.parse((await (await kv()).getItem(KEY)) ?? "[]") as FilmRecord[];
  } catch {
    return [];
  }
}

export async function saveRecord(rec: FilmRecord): Promise<void> {
  const all = await readAll();
  const next = [rec, ...all.filter((r) => r.id !== rec.id)].sort((a, b) => b.created - a.created);
  await (await kv()).setItem(KEY, JSON.stringify(next));
}

export async function loadRecord(id: string): Promise<FilmRecord | null> {
  return (await readAll()).find((r) => r.id === id) ?? null;
}

export async function allRecords(): Promise<FilmRecord[]> {
  return readAll();
}
```

```ts
import { BillingError, billingRun, billingUpload } from "@/src/lib/billing";

export class FalError extends Error {
  status?: number;
  constructor(message: string, status?: number) {
    super(message);
    this.status = status;
  }
}

export const isKeyError = (e: unknown) => e instanceof FalError && (e.status === 401 || e.status === 403);

export async function run<T = any>(app: string, input: Record<string, unknown>, _retries = 1): Promise<T> {
  try {
    return await billingRun<T>(app, input);
  } catch (e) {
    if (e instanceof BillingError) throw new FalError(e.message, e.status);
    throw e;
  }
}

export async function duration(url: string): Promise<number> {
  const r = await run<{ media: { duration: number } }>("fal-ai/ffmpeg-api/metadata", { media_url: url });
  return Number(r.media.duration);
}

export async function upload(file: { uri: string; name: string; type: string }): Promise<string> {
  return billingUpload(file);
}
```

```ts
export async function billingRun<T>(app: string, input: Record<string, unknown>): Promise<T> {
  if (!jobId) throw new BillingError(409, "no film in production");
  const s = await call(`/jobs/${jobId}/calls`, {
    method: "POST",
    headers: { "content-type": "application/json" },
    body: JSON.stringify({ app, input }),
  });
  const sub = (await s.json()) as { request_id?: string; error?: string };
  if (!s.ok || !sub.request_id) throw new BillingError(s.status, `${app}: ${sub.error ?? s.status}`, sub.error);
  const rid = sub.request_id;
  for (let i = 0; i < 40; i++) {
    const st = await call(`/jobs/${jobId}/calls/${rid}/status`);
    if (!st.ok) throw new BillingError(st.status, `${app}: status ${st.status}`);
    const body = (await st.json()) as { status?: string };
    if (body.status === "COMPLETED") break;
    if (body.status === "ERROR") throw new BillingError(500, `${app}: request failed`);
    if (i === 39) throw new BillingError(504, `${app}: timed out`);
    await sleep(Math.min(15000, 1500 * (i + 1)));
  }
  const r = await call(`/jobs/${jobId}/calls/${rid}/result`);
  if (!r.ok) throw new BillingError(r.status, `${app}: result failed`);
  return (await r.json()) as T;
}
```

```ts
      const done = await runFilm(rec, (e: JobEvent) => {
        setLastEvent(e.msg);
        const ui = mapEventStage(e.stage);
        if (ui && ui !== "done" && ui !== "error") setActiveStage(ui);
      });
      if (done.status === "done" && done.result.video) {
        const s = await settleFilm("done", done.result.video);
        setCredits(s.creditsLeft ?? Math.max(0, credits - cost));
        setRecord(done);
        setPhase("result");
      } else {
        throw new Error(done.error || "the film could not be finished");
      }
    } catch (e) {
      const s = await settleFilm("error").catch(() => ({ refund: 0 }));
      setRefund(s.refund ?? 0);
      setErrorText(String((e as Error).message ?? e));
      setPhase("error");
    }
  }

  async function resume(rec: FilmRecord) {
    if (rec.job_id) adoptJob(rec.job_id);
    setRecord(rec);
    setResumable(null);
    setPhase("progress");
    await finish(rec, 0);
  }
```

```ts
export function recordToFilm(rec: FilmRecord): Film | null {
  const r = rec.result ?? {};
  if (rec.status !== "done" || !r.video) return null;
  return {
    id: rec.id,
    title: rec.title || "Untitled film",

    narrator: rec.narrator?.name ?? "Narrator",
    styleLabel: rec.style_label || rec.style,

    generated: true,
    poster: FILMS[0].poster,

}

export async function generatedFilms(): Promise<Film[]> {
  return (await allRecords()).map(recordToFilm).filter((f): f is Film => Boolean(f));
```
